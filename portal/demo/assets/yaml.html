# The entire blueprint — left half

formatVersion: 1

inputs:
  # four placement choices
  os:      { type: string }
  size:    { type: string }
  cluster: { type: string }
  zone:    { type: string }

  # every custom attribute, in one object
  attributes:
    type: object
    default: {}
# right half
resources:
  Cloud_vSphere_Machine_1:
    type: Cloud.vSphere.Machine
    properties:
      image:  ${input.os}
      flavor: ${input.size}
      attributes: ${input.attributes}
      constraints:
        - tag: cluster:${input.cluster}
      networks:
        - network: ${resource...id}
          assignment: static

  Cloud_vSphere_Network_1:
    type: Cloud.vSphere.Network
    properties:
      networkType: existing
      constraints:
        - tag: zone:${input.zone}
That is the whole blueprint. Adding a custom attribute later changes nothing in this file.
